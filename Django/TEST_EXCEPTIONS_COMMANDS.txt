6. USAGE IN VIEWS (Example integration)
========================================

Add to detector/views.py:

  from django.http import JsonResponse
  from detector.exceptions import InvalidImageError, PredictionError, SessionError
  from detector.inference import predict_two_stage
  
  def predict_api(request):
      if request.method != 'POST':
          return JsonResponse({'error': 'POST required'}, status=405)
      
      try:
          if 'image' not in request.FILES:
              raise InvalidImageError("Image file required")
          
          image_file = request.FILES['image']
          
          # Validate image
          if image_file.size > 10 * 1024 * 1024:  # 10MB limit
              raise InvalidImageError("Image exceeds 10MB limit")
          
          # Run prediction (uses exceptions from detector.exceptions)
          try:
              prediction = predict_two_stage(image_file)
          except PredictionError as e:
              return JsonResponse({'error': f'Prediction failed: {str(e)}'}, status=500)
          
          # Save to database
          session_key = get_session_key(request)
          ScanHistory.objects.create(
              session_key=session_key,
              label=prediction['health_scan']['label'],
              confidence=prediction['health_scan']['confidence'],
              result=prediction
          )
          
          return JsonResponse(prediction)
      
      except InvalidImageError as e:
          return JsonResponse({'error': str(e)}, status=400)
      except SessionError as e:
          return JsonResponse({'error': 'Session error'}, status=500)
